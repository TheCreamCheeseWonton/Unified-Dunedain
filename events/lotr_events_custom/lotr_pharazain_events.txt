pharazain_events.5 = { # Event for Ray
	content_source = realms_dlc
	type = character_event
	title = pharazain_events.5.t
	desc = pharazain_events.5.desc
	theme = diplomacy

	override_background = { reference = bp1_courtyard_indian }
	
	immediate = {
		title:e_pharazain.holder = { save_scope_as = pharazain_holder  }
	}
	
	left_portrait = {
		character = root
		animation = stress
	}

	option = { # Become a trib of the Pharazain
		name = pharazain_events.5.a
		
		start_tributary_interaction_effect = {
			TRIBUTARY = root
			SUZERAIN = title:e_pharazain.holder
		}
		
		title:e_pharazain.holder = { 
			trigger_event = {
				id = pharazain_events.6
				days = 1
			}
		}

		ai_chance = {
			base = 70
			
			modifier = { # Game Rule
				has_game_rule = default_ai_behavior
				add = 0
			}

			modifier = { # Game Rule
				has_game_rule = weighted_ai_behavior
				add = 0
			}

			modifier = { # Game Rule
				has_game_rule = random_ai_behavior
				add = 10
			}
		}
	}

	option = { # Stay independent
		name = pharazain_events.5.b

		title:e_pharazain.holder = { 
			trigger_event = {
				id = pharazain_events.7
				days = 30 
			}
		}

		ai_chance = {
			base = 35 # LotR: This should be tied to WotR mode.

			modifier = { # Game Rule
				has_game_rule = default_ai_behavior
				add = 0
			}

			modifier = { # Game Rule
				has_game_rule = weighted_ai_behavior
				add = 15
			}

			modifier = { # Game Rule
				has_game_rule = random_ai_behavior
				add = 30
			}
		}
	}
}
