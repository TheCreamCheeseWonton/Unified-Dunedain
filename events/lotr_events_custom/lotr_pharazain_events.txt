pharazain_events.10 = {
	type = character_event
	title = pharazain_events.10.t
	desc = pharazain_events.10.desc
	theme = battle
	override_background = { reference = battlefield }
	
	trigger = { 
		has_title = title:e_pharazain
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:separatism_has_started
		}
		title:e_pharazain.holder = {
			any_vassal = {
				culture = { 
					NOT = { has_same_culture_heritage = title:e_pharazain.holder.culture }
				}
				opinion = {
					target = title:e_pharazain.holder
					value <= 10
				}
			}
		}
	}

	left_portrait = {
		character = root
		animation = disbelief
	}
	immediate = {
		title:e_pharazain = { save_scope_as = pharazain_title }
		scope:pharazain_title.holder ?= { save_scope_as = high_king_of_pharazain }
		#scope:pharazain_title.holder.culture ?= { save_scope_as = high_king_of_pharazain_culture }

		scope:high_king_of_pharazain = {
			random_vassal = {
				limit = {
					culture = { 
						NOT = { has_same_culture_heritage = title:e_pharazain.holder.culture }
					}
					opinion = {
						target = title:e_pharazain.holder
						value <= 10
					}
				}
				save_scope_as = leader_of_rebellion
			}
		}
	}

	option = { # (fight against rebelious/religious people)
		name = pharazain_events.10.a

		scope:leader_of_rebellion = {
			start_war = {
				casus_belli = independence_war
				target = scope:high_king_of_pharazain
				target_title = scope:high_king_of_pharazain.primary_title
			}
			every_character_war = {
				limit = { using_cb = independence_war } # In-case the random head of the attackers has other wars hes currently engaged in
				save_scope_as = war_against_pharazain
			}
			spawn_army = {
				name = bp2_yearly_6203_army_name
				levies = 600
				men_at_arms = {
					type = light_footmen
					stacks = 2
				}
				inheritable = no
				uses_supply = no
				location = scope:leader_of_rebellion.capital_province
			}
		}
	}
}
