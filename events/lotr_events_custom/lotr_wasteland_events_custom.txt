				compare_modifier = {
					value = scope:duel_value
					multiplier = -2
					min = -25
				}
				desc = wastelands_custom.001.a.duel_failure
				send_interface_toast = {
					left_icon = root
					title = wastelands_custom.001.failure.toast.t
					hidden_effect = {
						scope:obstacle_province = { 
							if = {
								limit = { has_building = ruins_of_arnor_bad_cleared }
								remove_building = ruins_of_arnor_bad_cleared
								add_to_global_variable_list = {
									name = unavailable_unique_decisions
									target = flag:province_is_bad_arnor_ruins
								}
							}
							else = { 
								remove_building = ruins_of_arnor_cleared
							}
						}
					}
					if = { 
						limit = { 
							is_target_in_global_variable_list = {
								name = unavailable_unique_decisions
								target = flag:province_is_bad_arnor_ruins
							}
						}
						scope:obstacle_province = { add_building = ruins_of_arnor_bad }
						scope:obstacle_province = { add_building = bandits }
						add_prestige = minor_prestige_loss
						remove_list_global_variable = {
							name = unavailable_unique_decisions
							target = flag:province_is_bad_arnor_ruins
						}
					}
					else = { 
						scope:obstacle_province = { add_building = ruins_of_arnor }
						scope:obstacle_province = { add_building = bandits }
						add_prestige = minor_prestige_loss
					}
					remove_short_term_gold = 60
				}
