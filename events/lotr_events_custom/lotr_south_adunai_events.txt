south_adunai_events.4 = { # Tanturaki Reformation Flavor
	content_source = realms_dlc
	type = character_event
	title = south_adunai_events.4.t
	desc = south_adunai_events.4.desc
	theme = stewardship_duty_focus
	override_background = { reference = bp1_courtyard_indian }

	left_portrait = {
		character = root
		animation = steward
	}

	option = { #Be Tanturaki
		name = south_adunai_events.4.a
		trigger = { NOT = { culture = { has_cultural_pillar = heritage_unified_adunaic } } }
		title:k_tanturak = { 
			every_de_jure_county = {
				if = { 
					limit = { culture = { has_cultural_pillar = heritage_southern_adunaic } NOT = { OR = { title:c_sarul title:c_ardinak title:c_bata_an_sarul title:c_azra_zhe } } }
					set_county_culture = culture:tanturaki
				}
			}
		}
		title:c_sarul = {
			set_county_culture = culture:tanturaki
			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20
			}
		}
		title:c_bata_an_sarul = {
			set_county_culture = culture:tanturaki
			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20
			}
		}
		title:c_azra_zhe = {
			set_county_culture = culture:tanturaki
			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20
			}
		}
		# Convert your family
		if = {
			limit = { 
				any_spouse = { 
					is_landed = no 
					culture = root.culture
				} 
			}
			every_spouse = {
				limit = {
					is_landed = no
					culture = root.culture
				}
				if = {
					limit = {
						culture != culture:tanturaki
					}
					set_culture = culture:tanturaki
				}
			}
		}
		if = {
			limit = {
				any_close_family_member = {
					is_landed = no
					NOT = { is_spouse_of = root }
					culture = root.culture
				}
			}
			every_close_family_member = {
				limit = {
					is_landed = no
					NOT = { is_spouse_of = root }
					culture = root.culture
				}
				custom = all_unlanded_family_members
				if = {
					limit = {
						culture != culture:tanturaki
					}
					set_culture = culture:tanturaki
				}
			}
		}
		if = {
			limit = {
				root.culture != culture:tanturaki
			}
			set_culture = culture:tanturaki
		}
		add_legitimacy = 50
		add_character_modifier = {
			modifier = lotr_converting_tanturak_modifier
			years = 10
		}
	}
	
	option = { #Not be Tanturaki
		name = south_adunai_events.4.b
		trigger = { NOT = { culture = { has_cultural_pillar = heritage_unified_adunaic } } }
		title:k_tanturak = { 
			every_de_jure_county = {
				if = { 
					limit = { culture = { has_cultural_pillar = heritage_southern_adunaic } NOT = { OR = { title:c_sarul title:c_ardinak title:c_bata_an_sarul title:c_azra_zhe } } }
					set_county_culture = culture:tanturaki
				}
			}
		}
		title:c_sarul = {
			set_county_culture = culture:tanturaki
			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20
			}
		}
		title:c_bata_an_sarul = {
			set_county_culture = culture:tanturaki
			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20
			}
		}
		title:c_azra_zhe = {
			set_county_culture = culture:tanturaki
			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20
			}
		}
		add_legitimacy = 50
		add_character_modifier = {
			modifier = lotr_converting_tanturak_modifier
			years = 10
		}
	}
	
	option = { #Super not be Tanturaki
		name = south_adunai_events.4.c
		title:c_sarul = {
			set_county_culture = root.culture
			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20
			}
		}
		title:c_bata_an_sarul = {
			set_county_culture = root.culture
			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20
			}
		}
		title:c_azra_zhe = {
			set_county_culture = root.culture
			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20
			}
		}
		add_legitimacy = 50
		add_character_modifier = {
			modifier = lotr_converting_tanturak_modifier
			years = 10
		}
	}
}
