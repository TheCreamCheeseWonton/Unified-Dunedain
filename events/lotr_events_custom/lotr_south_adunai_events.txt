south_adunai_events.4 = { # Tanturaki Reformation Flavor
	content_source = realms_dlc
	type = character_event
	title = south_adunai_events.4.t
	desc = south_adunai_events.4.desc
	theme = stewardship_duty_focus

	left_portrait = {
		character = root

	option = { #Be Tanturaki
		name = south_adunai_events.4.a
		trigger = { NOT = { culture = { has_cultural_pillar = heritage_unified_adunaic } } }
		title:k_tanturak = { 
			every_de_jure_county = {
				if = { 
					limit = { culture = { has_cultural_pillar = heritage_southern_adunaic } NOT = { OR = { title:c_sarul title:c_ardinak title:c_bata_an_sarul title:c_azra_zhe } } }
					set_county_culture = culture:tanturaki
				}
			}
		}
		title:c_sarul = {
			set_county_culture = culture:tanturaki
			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20
			}
		}
		title:c_bata_an_sarul = {
			set_county_culture = culture:tanturaki
			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20
			}
		}
		title:c_azra_zhe = {
			set_county_culture = culture:tanturaki
			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20
			}
		}
		# Convert your family
		if = {
			limit = { 
				any_spouse = { 
					is_landed = no 
					culture = root.culture
				} 
			}
			every_spouse = {
				limit = {
					is_landed = no
					culture = root.culture
				}
				if = {
					limit = {
						culture != culture:tanturaki
					}
					set_culture = culture:tanturaki
				}
			}
		}
		if = {
			limit = {
				any_close_family_member = {
					is_landed = no
					NOT = { is_spouse_of = root }
					culture = root.culture
				}
			}
			every_close_family_member = {
				limit = {
					is_landed = no
					NOT = { is_spouse_of = root }
					culture = root.culture
				}
				custom = all_unlanded_family_members
				if = {
					limit = {
						culture != culture:tanturaki
					}
					set_culture = culture:tanturaki
				}
			}
		}
		if = {
			limit = {
				root.culture != culture:tanturaki
			}
			set_culture = culture:tanturaki
		}
		add_legitimacy = 50
		add_character_modifier = {
			modifier = lotr_converting_tanturak_modifier
			years = 10
		}
	}
	
	option = { #Not be Tanturaki
		name = south_adunai_events.4.b
		trigger = { NOT = { culture = { has_cultural_pillar = heritage_unified_adunaic } } }
		title:k_tanturak = { 
			every_de_jure_county = {
				if = {

					set_county_culture = culture:tanturaki
				}
			}
		}

			add_county_modifier = {
				modifier = lotr_converting_tanturak_county_modifier
				years = 20

		}

south_adunai_events.2000 = { # Invite elves to Baranagath
	content_source = realms_dlc
	type = character_event
	title = south_adunai_events.2000.t
	desc = south_adunai_events.2000.desc
	theme = elven

	override_background = {
		reference = elven_holding
	}
	
	right_portrait = {
		character = scope:elf_cuind
		animation = personality_honorable
	}

	immediate = {
		hidden_effect = { # Create elf
			if = { # Cuind
				limit = { 
					character:lineofmeglasul9 = { 
						is_alive = yes
					}
					character:lineofmeglasul9 = {
						is_landed = no
					}
				}
				character:lineofmeglasul9 = {
					save_scope_as = elf_cuind
				}
			}
			else_if = {
				limit = { 
					character:lineofmeglasul7 = { 
						is_alive = yes
					}
					character:lineofmeglasul7 = {
						is_landed = no
					}
				}
				character:lineofmeglasul7 = {
					save_scope_as = elf_cuind
				}
			
			}
			else = { 
				create_character = { # Cuind Elf
					location = root.capital_province
					culture = culture:cuind
					faith = faith:faith_cuind
					age = { 100 6950 }
					#random_traits = yes
					trait = education_stewardship_4

					trait = patient

					trait = lifestyle_poet

					diplomacy = { 8 10 }
					stewardship = { 6 14 }
					martial = { 6 8 }
					intrigue = { 10 12 }
					learning = { 6 8 }
					prowess = { 6 8 }

					gender_female_chance = 50
					save_scope_as = elf_cuind
					after_creation = {
						scope:elf_cuind = { replace_unelven_traits = yes }
					}
				}
			}
		}

		title:c_baranagath = { # Changes province to elven holding
			every_county_province = {
				limit = {
					has_holding = yes
				}
				set_holding_type = elven_holding
			}
		}
		#title:c_edhellond_2 = { # Changes province to elven holding
		#	every_county_province = {
		#		limit = {
		#			has_holding = yes
		#		}
		#		set_holding_type = elven_holding
		#	}
		#}
	}

	option = { # Invite Elves from Cuind
		name = south_adunai_events.2000.a
		custom_tooltip = south_adunai_events.2000.a_tt
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:c_baranagath = {
			change_title_holder = {
				holder = scope:elf_cuind
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change

		title:c_baranagath = {
			set_county_culture = scope:elf_cuind.culture
			set_county_faith = scope:elf_cuind.faith
		}


		scope:elf_cuind = {
			add_opinion = {
				target = root
				modifier = invite_elves_cuind
				years = 25
			}
			hidden_effect = {
				add_gold = 150
				add_prestige = 750
				add_piety = 500
			}
		}

		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:d_cinta_toina = {
			change_title_holder = {
				holder = scope:elf_cuind
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change

		apply_elven_influence_1 = yes
		upgrade_baranagath = yes
	}

	after = {
		trigger_event = south_adunai_events.2001

		# Some elven characters to make sure the elf has some nice courtiers
		create_character = {
			employer = title:d_cinta_toina.holder
			faith = title:d_cinta_toina.holder.faith
			culture = title:d_cinta_toina.holder.culture
			gender_female_chance = 50
			age = { 100 6950 }
			random_traits = yes
		}
		create_character = {
			employer = title:d_cinta_toina.holder
			faith = title:d_cinta_toina.holder.faith
			culture = title:d_cinta_toina.holder.culture
			gender_female_chance = 50
			age = { 100 6950 }
			random_traits = yes
		}
		create_character = {
			employer = title:d_cinta_toina.holder
			faith = title:d_cinta_toina.holder.faith
			culture = title:d_cinta_toina.holder.culture
			gender_female_chance = 50
			age = { 100 6950 }
			random_traits = yes
		}
		create_character = {
			employer = title:d_cinta_toina.holder
			faith = title:d_cinta_toina.holder.faith
			culture = title:d_cinta_toina.holder.culture
			gender_female_chance = 50
			age = { 100 6950 }
			random_traits = yes
		}
		create_character = {
			employer = title:d_cinta_toina.holder
			faith = title:d_cinta_toina.holder.faith
			culture = title:d_cinta_toina.holder.culture
			gender_female_chance = 50
			age = { 100 6950 }
			random_traits = yes
		}

		# De jure and vassal shifts to make sure the Elves don't get caught in a revocation cycle
		hidden_effect = {
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			title:d_cinta_toina.holder = {
				change_liege = {
					liege = root
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
			
			# Get rid of any remaining human claims 
			title:d_cinta_toina = {
				every_claimant = {
					limit = { is_human = yes }
					remove_claim = title:d_cinta_toina 
				}
			}
			title:c_baranagath = {
				every_claimant = {
					limit = { is_human = yes }
					remove_claim = title:c_baranagath 
				}
			}
		}
	}
}
