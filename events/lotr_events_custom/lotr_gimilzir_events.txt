gimilzir_events.701 = { # Inzilloni Contract
	content_source = realms_dlc
	type = character_event
	title = gimilzir_events.701.t
	desc = gimilzir_events.701.desc
	theme = laamp
	#window = visit_settlement_window
	override_background = {
		reference = throne_room_mediterranean
	}
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:inzilloniholder
		animation = debating
	}
	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}
	immediate = {
		title:c_inzilloni.holder = { save_scope_as = inzilloniholder }
		remove_character_flag = gimilzir_tanturak_contract_inzilloni
	}
	option = { # Try to convince him
		name = gimilzir_events.701.a
		domicile ?= {
			change_provisions = medium_provisions_gain
			move_domicile = title:c_inzilloni.title_province
		}
		duel = {
 			skill = diplomacy
 			target = scope:inzilloniholder
 			40 = {
				# You convince him
 				desc = gimilzir_events.701.a.success
 				compare_modifier = {
 					value = scope:duel_value
 					multiplier = 3.5
 					min = -49
 				}
 				send_interface_toast = {
 					title = gimilzir_events.701.a.success
 					left_icon = root
 					right_icon = scope:inzilloniholder
 				}
				scope:inzilloniholder = {
					add_opinion = {
						target = root
						modifier = pleased_opinion
						opinion = 15
					}
				}
				trigger_event = gimilzir_events.702
				add_prestige = 25
 			}
 			60 = {
 				# You fail to convince him
 				desc = gimilzir_events.701.a.failure
 				compare_modifier = {
 					value = scope:duel_value
 					multiplier = -3.5
 					min = -49
 				}
 				send_interface_toast = {
 					title = gimilzir_events.701.a.failure
 					left_icon = root
 					right_icon = scope:inzilloniholder
 				}
				scope:inzilloniholder = {
					add_opinion = {
						target = root
						modifier = disappointed_opinion
						opinion = -15
					}
				}
				add_prestige = -25
				trigger_event = gimilzir_events.703
 			}
 		}
	}
	
	option = { # You pay him
		name = gimilzir_events.701.b
		domicile ?= {
			change_provisions = medium_provisions_gain
			move_domicile = title:c_inzilloni.title_province
		}
		custom_tooltip = gimilzir_events.701.a.tt
		pay_reserved_gold = { target = scope:inzilloniholder gold = 150 }
		trigger_event = gimilzir_events.702
	}
	
	option = { # Reject his offer and do it sneaky like
		name = gimilzir_events.701.c
		ai_chance = { base = 100 }
		custom_tooltip = gimilzir_events.701.b.tt
		trigger_event = {
			id = gimilzir_events.704
			days = 10
		}
	}
	
	option = { # Reject his offer and fuck off
		name = gimilzir_events.701.d
		ai_chance = { base = 100 }
		random_character_active_contract = {
			limit = { has_task_contract_type = lotr_gimilzir_tanturak_contract_inzilloni }
			complete_task_contract = failure_standard
		}
		custom_tooltip = gimilzir_events.701.c.tt
		trigger_event = {
			id = gimilzir_events.710 
			days = 20
		}
	}
}
