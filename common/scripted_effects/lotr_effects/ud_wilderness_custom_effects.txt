##Wilderness Submod
create_artifact_palantir_romenna_effect = {
	# Get the character the artifact is being made for.
	$OWNER$ = { save_scope_as = owner }
	# Not really used, but if we don't set the scopes we get errors in the feature selection
	set_artifact_rarity_illustrious = yes

	# Create the artifact
	create_artifact = {
		name = arnorian_figurine_name
		description = arnorian_figurine_description
		type = pedestal
		visuals = numenorean_figurine
		template = general_unique_template
		wealth = scope:wealth
		quality = scope:quality
		modifier = arnorian_figurine_modifier
		# template = palantir_template
		history = {
			type = created
			date = 2.2.1
			actor = character:lineoffinwe3 #Fëanor
			recipient = character:lineoffinwe3 #Fëanor
			location = province:b_tirion #Tirion
		}
		modifier = palantir_modifier
		save_scope_as = newly_created_artifact
		decaying = no
	}

	scope:newly_created_artifact = {
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = palantir_minasanor
		set_variable = palantir
		set_variable = ai_ungiftable
		save_scope_as = epic

		add_artifact_title_history = {
		target = title:e_numenor
		date = 1063.1.1
		}
	}
}
