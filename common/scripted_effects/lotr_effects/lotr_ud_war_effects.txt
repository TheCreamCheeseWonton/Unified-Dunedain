###
# GEOGRAPHICAL_REGION 	--> Area in which the invading party appears
# $GRAB_LANDING_COUNTY_EFFECT$ = yes 	--> 
#							GRAB_LANDING_COUNTY_EFFECT => The effect which grabs the specific landing county. 
#									 Needs to save the county to a scope called landing_county
#
# $REVIVE_DEAD_CHARACTER_EFFECT$ = yes		-->
#							REVIVE_DEAD_CHARACTER_EFFECT => The effect which grabs the character that needs to be revived
#
# $INVASION_LANDING_EFFECT$ = yes		-->
#							INVASION_LANDING_EFFECT => Additional scripted effect which can be used
#
# $START_INVASION_WAR_EFFECT$ = yes		-->
#							START_INVASION_WAR_EFFECT => Additional scripted effect which can be used
#
#
# NOTES: 	If there are no scripted effects the coder wants to add, simply do START_INVASION_WAR_EFFECT = "#" (WITH the quotation marks)
#			This will effectively comment out that line.
###

civil_war_start_general_effect = {
	if = { #Bellakar
		limit = { geographical_region:$GEOGRAPHICAL_REGION$ = geographical_region:middleearth_harad_far_bellakar }
		civil_war_spawn_bellakar_character = yes
	}
	else_if = { #Thani Hazad
		limit = { geographical_region:$GEOGRAPHICAL_REGION$ = geographical_region:custom_mannish_thani_hazad }
		civil_war_spawn_thani_hazad_character = yes
	}
	else_if = { #Anballukhor
		limit = { geographical_region:$GEOGRAPHICAL_REGION$ = geographical_region:special_k_an_balkumagan }
		civil_war_spawn_anballukhor_character = yes
	}

	### Grab county at which conqueror will land at. 
	### Needs to save the county to a scope called landing_county
	$GRAB_LANDING_COUNTY_EFFECT$ = yes
	
	### Effect that can be fired to re-create a character, such as Pharazon
	$REVIVE_DEAD_CHARACTER_EFFECT$ = yes

	debug_log = "SUNSET INVASION SCOPES:"
	debug_log_scopes = yes

	scope:invader_char = {
		$INVASION_LANDING_EFFECT$ = yes
		$START_INVASION_WAR_EFFECT$ = yes
	}
}

civil_war_defender_looses_initial_revolt_war_effect = {
	if = {
		limit = { # Major News
			NOT = { has_game_rule = no_news } 
			scope:defender = {
				has_title = title:h_pharazain
			}
		}
		every_player = { #Global News Announcement
			trigger_event = pharazain_events.28
		}
	}
	scope:attacker = {
		destroy_title = title:h_pharazain
	}
}

civil_war_defender_wins_initial_revolt_war_effect = {
	scope:attacker = {
		every_held_county = {
			add_to_list = transfer_titles
		}
	}

	create_title_and_vassal_change = {
		type = conquest
		save_scope_as = change
		add_claim_on_loss = yes
	}
	
	every_in_list = {
		list = transfer_titles
		change_title_holder = {
			holder = scope:defender
			change = scope:change
		}
	}
	scope:attacker.capital_county = {
		every_county_province = {
			remove_province_modifier = $PROVINCE_MODIFIER$
		}
	}
	if = { 
		limit = { 
			scope:defender = { 
				NOT = { 
					any_character_war = { 
						count > 1
						using_cb = lotr_pharazain_revolt_invasion_cb 
					}
				}
			}
		}
		scope:defender = {
			trigger_event = { 
				id = pharazain_events.27
				days = 7
			}
		}
	}
		
	resolve_title_and_vassal_change = scope:change
}

ud_civil_war_general_landing_effect = {
	create_title_and_vassal_change = {
		type = returned
		save_scope_as = change
	}
	scope:landing_county ?= {
		change_title_holder = {
			holder = scope:invader_char
			change = scope:change
			take_baronies = yes
		}
	}
	resolve_title_and_vassal_change = scope:change

	#scope:landing_county = {
	#	every_county_province = {
	#		limit = { 
	#			NOT = { has_holding_type = castle_holding }
	#		}
	#		set_holding_type = castle_holding
	#	}
	#}
	scope:landing_county ?= {
		every_county_province = {

			add_province_modifier = {
				modifier = $PROVINCE_MODIFIER$
				years = 10
			}
		}
	}
	
	if = {
		limit ?= { scope:invader_char.top_liege != scope:invader_char }
		scope:invader_char ?= {
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = independent_change
			}
			becomes_independent = {
				change = scope:independent_change
			}
			resolve_title_and_vassal_change = scope:independent_change
		}
	}

	if = {
		limit = { government_has_flag = government_is_wastelands }
		scope:invader_char = { change_government = $GOVERNMENT$ }
	}

	scope:invader_char ?= { get_title = $TITLE_TO_SPAWN_WITH$ }
	
	scope:invader_char = {
		if = {
			limit = { is_landed = yes }
			add_legitimacy = 3000
		}
		add_prestige = 5000
		add_piety = 5000
		add_gold = 1500
		#set_variable = invasion_cb_unlock
	}

	scope:landing_county = {
		#set_county_faith = $LANDING_COUNTY_FAITH$
		#set_county_culture = $LANDING_COUNTY_CULTURE$
	
		if = {
			limit = { has_county_modifier = marauding_orc_dregs_modifier }
			remove_county_modifier = marauding_orc_dregs_modifier
		}
		if = {
			limit = { has_county_modifier = hidden_elven_refugees_modifier }
			remove_county_modifier = hidden_elven_refugees_modifier
		}
	}

	scope:invader_char = {
		every_courtier_or_guest = {
			limit = { is_wastelands = yes }
			silent_disappearance_effect = yes
		}
	}
}
