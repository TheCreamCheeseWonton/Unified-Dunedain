has_dunedain_court_illustration_trigger = { # Gondorian
	primary_title != title:e_gondor_steward
	# capital_county.culture = {
	# 	AND = {
	# 		has_cultural_pillar = heritage_dunedain
	# 		has_cultural_tradition = tradition_gondorian_legacy 
	# 	}
	# }
	trigger_if = {
		limit = { exists = capital_province }
		AND = {
			capital_province.culture = { has_cultural_pillar = heritage_dunedain }
			OR = {
				capital_province = { geographical_region = middleearth_west_gondor }
				capital_province = { geographical_region = custom_gondor_calenardhon }
				capital_province = { geographical_region = middleearth_mordor }
				capital_province = { geographical_region = middleearth_harad_near_harondor }
			}
		}
	}
	trigger_else_if = {
		limit = { NOT = { exists = capital_province } }
		trigger_if = { 
			limit = { exists = host.capital_province }
			AND = {
				OR = { 
					host.capital_province.culture = { has_cultural_pillar = heritage_dunedain }
					host.capital_province.culture = { has_cultural_pillar = heritage_unified_adunaic }
				}
				OR = {
					host.capital_province = { geographical_region = middleearth_west_gondor }
				}
			}
		}
		trigger_else_if = {
			limit = { exists = liege.capital_province }
			AND = {
				OR = { 
					liege.capital_province.culture = { has_cultural_pillar = heritage_dunedain }
					liege.capital_province.culture = { has_cultural_pillar = heritage_unified_adunaic }
				}
				OR = {
					liege.capital_province = { geographical_region = middleearth_west_gondor }
				}
			}
		}
		trigger_else = { always = yes }
	}
	trigger_else = { always = yes }
}
