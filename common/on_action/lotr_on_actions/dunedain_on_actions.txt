	trigger = {
		has_title = title:e_pharazain
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:separatism_has_started
		}
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:decision_separatism_is_finished
			}
		}
	}

	random_events = {
		chance_to_happen = 33
		33 = pharazain_events.10

dunedain_yearly_global_pulse = { 
	effect = { 
		every_county ?= {
			limit = { 
				OR = { 
					has_county_modifier = custom_harder_to_colonize_elven
					has_county_modifier = custom_harder_to_colonize_dwarven
					has_county_modifier = custom_harder_to_colonize_dunedain
				}
				is_wastelands = no
			}
			remove_county_modifier = custom_harder_to_colonize_elven
			remove_county_modifier = custom_harder_to_colonize_dwarven
			remove_county_modifier = custom_harder_to_colonize_dunedain
		}
	}
}
