			deactivate_holy_site = sarul_faithful
			deactivate_holy_site = ardinak_faithful

		}
		##Quonton
		character:linegimilzadan2 = {
			if = {
				limit = {
					is_alive = yes
				}
				lotr_set_lore_character_for_bookmark = { TYPE_OF_CONTENT = story_content }
			}
		}
		character:linegimilethtwocustom2 = {
			if = { 
				limit = {
					is_alive = yes
				}
				start_dog_story_cycle_effect = yes
				assign_dog_gender_effect = { GENDER = female }
				assign_dog_fur_color_effect = { COLOR = brown }
				assign_dog_eye_color_effect = { COLOR = brown }
			}
		}

		##Bad Arnorian Ruins
		province:4415 = { if = { limit = { is_wastelands = yes } add_province_modifier = { modifier = util_wasteland_3 days = 0 } add_building = ruins_of_arnor_bad } }
		province:4714 = { if = { limit = { is_wastelands = yes } add_province_modifier = { modifier = util_wasteland_3 days = 0 } add_building = ruins_of_arnor_bad } }

			add_building = ruins_of_arnor
		}

dunedain_yearly_global_pulse = { 
	effect = { 
		every_county ?= {
			limit = { 
				OR = { 
					has_county_modifier = custom_harder_to_colonize_elven
					has_county_modifier = custom_harder_to_colonize_dwarven
					has_county_modifier = custom_harder_to_colonize_dunedain
				}
				is_wastelands = no
			}
			remove_county_modifier = custom_harder_to_colonize_elven
			remove_county_modifier = custom_harder_to_colonize_dwarven
			remove_county_modifier = custom_harder_to_colonize_dunedain
		}
		if = { 
			limit = { 
				character:linegimilzadan2 = {
					piety_level >= 5
					NOT = { 
						is_target_in_global_variable_list = {
							name = unavailable_unique_events
							target = flag:gimilzir_saintly_life
						}
					}
				}
			}
			character:linegimilzadan2 = {
				trigger_event = { id = gimilzir_events.33 years = { 3 7 } }
			}
		}
	}
}

dunedain_on_vassal_gained = {
	effect = {
		#pharazain_debuffs
		#trigger_event = pharazain_events.10
		if = {
			limit = { 
				NOT = { has_game_rule = default_evil_strength }
				has_game_rule = no_they_arent
				is_independent_ruler = yes 
				religion = { is_in_family = rf_forcesofevil }
				NOT = {
					OR = { 
						has_character_modifier = evil_weakened_modifier
						has_character_modifier = evil_strengthened_1_modifier
						has_character_modifier = evil_strengthened_2_modifier
						has_character_modifier = evil_strengthened_3_modifier
					}
				}
				NOT = { 
					this = character:lineofsauron 
				}
			} 
			switch = {
				trigger = has_game_rule
				weakened_evil_strength = { add_character_modifier = evil_weakened_modifier }
				strong_evil_strength = { add_character_modifier = evil_strengthened_1_modifier }
				very_strong_evil_strength = { add_character_modifier = evil_strengthened_2_modifier }
				apocalyptic_evil_strength = { add_character_modifier = evil_strengthened_3_modifier }
			}
		}
		if = {
			limit = { 
				NOT = { has_game_rule = default_evil_strength }
				has_game_rule = yes_they_are
				is_independent_ruler = yes
				OR = { 
					religion = { is_in_family = rf_forcesofevil } 
					faith = { has_doctrine = doctrine_dominion_of_the_dark }
				}
				NOT = {
					OR = { 
						has_character_modifier = evil_weakened_modifier
						has_character_modifier = evil_strengthened_1_modifier
						has_character_modifier = evil_strengthened_2_modifier
						has_character_modifier = evil_strengthened_3_modifier
					}
				}
				NOT = {
					OR = {
						has_character_modifier = sauron_strengthened_4_modifier
						has_character_modifier = sauron_strengthened_3_modifier
						has_character_modifier = sauron_strengthened_2_modifier
						has_character_modifier = sauron_strengthened_1_modifier
						has_character_modifier = sauron_weakened_modifier
					}
				}
				NOT = { 
					this = character:lineofsauron 
				}
			} 
			switch = {
				trigger = has_game_rule
				weakened_evil_strength = { add_character_modifier = evil_weakened_modifier }
				strong_evil_strength = { add_character_modifier = evil_strengthened_1_modifier }
				very_strong_evil_strength = { add_character_modifier = evil_strengthened_2_modifier }
				apocalyptic_evil_strength = { add_character_modifier = evil_strengthened_3_modifier }
			}
		}
	}
}
