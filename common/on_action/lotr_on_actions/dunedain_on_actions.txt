dunedain_on_vassal_gained = {
	effect = {
		if = {
			limit = { 
				NOT = { has_game_rule = default_evil_strength }
				has_game_rule = no_they_arent
				is_independent_ruler = yes 
				religion = { is_in_family = rf_forcesofevil }
				NOT = {
					OR = { 
						has_character_modifier = evil_weakened_modifier
						has_character_modifier = evil_strengthened_1_modifier
						has_character_modifier = evil_strengthened_2_modifier
						has_character_modifier = evil_strengthened_3_modifier
					}
				}
				NOT = { 
					this = character:lineofsauron 
				}
			} 
			switch = {
				trigger = has_game_rule
				weakened_evil_strength = { add_character_modifier = evil_weakened_modifier }
				strong_evil_strength = { add_character_modifier = evil_strengthened_1_modifier }
				very_strong_evil_strength = { add_character_modifier = evil_strengthened_2_modifier }
				apocalyptic_evil_strength = { add_character_modifier = evil_strengthened_3_modifier }
			}
		}
		if = {
			limit = { 
				NOT = { has_game_rule = default_evil_strength }
				has_game_rule = yes_they_are
				is_independent_ruler = yes
				OR = { 
					religion = { is_in_family = rf_forcesofevil } 
					faith = { has_doctrine = doctrine_dominion_of_the_dark }
				}
				NOT = {
					OR = { 
						has_character_modifier = evil_weakened_modifier
						has_character_modifier = evil_strengthened_1_modifier
						has_character_modifier = evil_strengthened_2_modifier
						has_character_modifier = evil_strengthened_3_modifier
					}
				}
				NOT = { 
					this = character:lineofsauron 
				}
			} 
			switch = {
				trigger = has_game_rule
				weakened_evil_strength = { add_character_modifier = evil_weakened_modifier }
				strong_evil_strength = { add_character_modifier = evil_strengthened_1_modifier }
				very_strong_evil_strength = { add_character_modifier = evil_strengthened_2_modifier }
				apocalyptic_evil_strength = { add_character_modifier = evil_strengthened_3_modifier }
			}
		}
	}
}
