	decision_group_type = realms_in_exile_major

		current_year >= 5630		# After it becomes dejure Umbar
		culture = {
			##Quonton
			OR = {

		}
		is_landed = yes

			has_title = title:e_bellakar
			has_title = title:h_pharazain

				name = unavailable_unique_decisions
				target = flag:reclaim_mardruak_decision
			}

decision_bellakar_x_gondor = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_arnor_01.dds"
	}
	title = decision_bellakar_x_gondor
	selection_tooltip = decision_bellakar_x_gondor_tooltip
	desc = decision_bellakar_x_gondor_desc
	confirm_text = decision_bellakar_x_gondor_confirm
	
	# SHOWN
	decision_group_type = realms_in_exile_major
	
	is_shown = {
		OR = {
			has_title = title:e_bellakar
			has_title = title:h_pharazain
		}
		NOT = {
			has_title = title:e_gondor
			has_title = title:h_reunited_kingdom
		}
		faith_is_good = yes
		OR = {
			AND = {
				exists = title:e_gondor.holder # Checks if aragorn has returned
				in_diplomatic_range = title:e_gondor.holder
			}
			AND = {
				exists = title:h_reunited_kingdom.holder
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:decision_solidify_reunited_kingdom
				}
				in_diplomatic_range = title:h_reunited_kingdom.holder
			}
		}
		dynasty = dynasty:dynasty_tumakveh
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:gondor_x_bellakar
			}
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:bellakar_x_gondor
			}
		}
		trigger_if = {
			limit = {
				exists = title:h_reunited_kingdom.holder
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:decision_solidify_reunited_kingdom
				}
			}
			NOT = {
				is_allied_to = title:h_reunited_kingdom.holder
			}
		}
		trigger_else = {
			trigger_if = {
				limit = {
					exists = title:e_gondor.holder
				}
				NOT = {
					is_allied_to = title:e_gondor.holder
				}
			}
		}
		global_var:bellakar_north_focus = no
	}

	# IS VALID
	is_valid = {
		prestige_level >= 4
		sauron_is_alive = yes
		trigger_if = {
			limit = {
				exists = title:h_reunited_kingdom.holder
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:decision_solidify_reunited_kingdom
				}
			}
			title:h_reunited_kingdom.holder = {
				faith_is_good = yes
				OR = {
					# dynasty = dynasty:dynasty_elros
					dynasty = dynasty:dynasty_elendil
					dynasty = dynasty:dynasty_anarion
				}
			}
		}
		trigger_else = {
			title:e_gondor.holder = {
				faith_is_good = yes
				OR = {
					# dynasty = dynasty:dynasty_elros
					dynasty = dynasty:dynasty_elendil
					dynasty = dynasty:dynasty_anarion
				}
			}
		}

		custom_tooltip = {
			text = decision_bellakar_x_gondor_custom_tooltip_a
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:reclaim_mardruak_decision
			}
		}
		custom_tooltip = {
			text = decision_bellakar_x_gondor_custom_tooltip_b
			OR = {
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:bellakar_felayan_rapprochement
				}
				# In case you do it not via the decision
				has_title = title:k_felaya
				any_vassal_or_below = {
					has_title = title:k_felaya
				}
				completely_controls = title:k_felaya
				##Quonton
				has_title = title:h_pharazain
			}
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_available_adult = yes
		is_at_war = no
	}

	#EFFECT
	effect = { # make an alliance between Gondor x Bellakar for next ~150 years or something idk
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:bellakar_x_gondor
		}
		if = {
			limit = {
				exists = title:h_reunited_kingdom.holder
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:decision_solidify_reunited_kingdom
				}
			}
			title:h_reunited_kingdom.holder = {
				trigger_event = {
					id = bellakar.0250
					days = { 15 30 }
				} 
			}
		}
		else = {
			title:e_gondor.holder = {
				trigger_event = {
					id = bellakar.0250
					days = { 15 30 }
				} 
			}
		}
	}

	# AI
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 24
		hegemony = 24
	}
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 100
	}
}

		hegemony = 60
	}
	sort_order = 40

	desc = anbalkumagan_grain_dole_decision_desc
	selection_tooltip = anbalkumagan_grain_dole_decision_tooltip
	decision_group_type = realms_in_exile_major

	is_shown = {
		has_ep3_dlc_trigger = yes

		# Empire tier Numenorean in the south, who isn't from AB itself
		highest_held_title_tier >= tier_empire
		NOT = { capital_province = { geographical_region = custom_anbalkumagan } } # This isn't strange for AB.
		capital_province = { geographical_region = special_pharazain }
		lotr_has_numenorean_culture_trigger = yes

		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:anbalkumagan_grain_dole_decision
			}
		}
	}

	is_valid = {
		# completely_controls = title:d_korlea # Should work for any capital.
		completely_controls_region = custom_anbalkumagan
		custom_tooltip = {
			text = enough_granaries_in_anbalkumagan
			any_county_in_region = {
				region = custom_anbalkumagan
				count >= 3
				any_county_province = {
					has_building_or_higher = cereal_fields_05
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_at_war = no
		NOT = { exists = involved_activity }
		is_imprisoned = no
		custom_tooltip = {
			NOT = {
				capital_county = { has_county_modifier = bread_and_circuses_anbalkumagan_county_modifier }
			}
			text = grain_dole_already_active_anbalkumagan
		}
	}

	cost = {
		gold = grain_dole_cost
	}

	effect = {
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:anbalkumagan_grain_dole_decision
		}
		trigger_event = bellakar.3000 
		reinstitute_grain_dole_decision_effect_anbalkumagan = yes
	}

	ai_potential = {
		OR = {
			has_title = title:e_bellakar
			has_title = title:e_thani_hazad
			has_title = title:h_pharazain
			has_title = title:h_numenor_in_exile

	}
