#Tanturak
k_tanturak = {
	item = {
		trigger = {
			coa_switch_to_adunai_trigger = yes
			OR = { 
				faith = { has_doctrine = tenet_adunaic_birthright }
				holder.religion = religion:sauronic_religion
			}
		}
		coat_of_arms = k_tanturak_adunaic
	}
	item = {
		trigger = {
			coa_switch_to_adunai_trigger = yes
			NOT = {

					holder.religion = religion:sauronic_religion
				}
			}
		}
		coat_of_arms = k_tanturak_adunaic_less_evil
	}
 	item = {
